package service

// Diagnostic only: count actual DML statements on synthetic PostgreSQL data.
// This is not a throughput benchmark and is not added to the product test suite.
import (
	"context"
	"os"
	"regexp"
	"strings"
	"sync"
	"testing"
	"time"

	"github.com/QuantumNous/new-api/common"
	"github.com/QuantumNous/new-api/model"
	relaycommon "github.com/QuantumNous/new-api/relay/common"
	"github.com/gin-gonic/gin"
	"github.com/stretchr/testify/assert"
	"github.com/stretchr/testify/require"
	"gorm.io/gorm"
	"gorm.io/gorm/logger"
)

type sqlWriteDiagnostic struct {
	logger.Interface
	mu     sync.Mutex
	counts map[string]int
}

var diagnosticDML = regexp.MustCompile("(?i)^(INSERT INTO|UPDATE|DELETE FROM)\\s+[\"`]?([a-z_]+)")

func (d *sqlWriteDiagnostic) Trace(_ context.Context, _ time.Time, fc func() (string, int64), err error) {
	sql, _ := fc()
	if m := diagnosticDML.FindStringSubmatch(strings.TrimSpace(sql)); len(m) > 0 && err == nil {
		d.mu.Lock()
		d.counts[strings.ToUpper(m[1])+" "+m[2]]++
		d.mu.Unlock()
	}
}
func (d *sqlWriteDiagnostic) flush(t *testing.T, phase string) {
	d.mu.Lock()
	defer d.mu.Unlock()
	t.Logf("DML phase=%s counts=%v", phase, d.counts)
	d.counts = map[string]int{}
}

func TestIsolatedSQLWriteDiagnostic(t *testing.T) {
	require.NotEmpty(t, os.Getenv("CILAI_TEST_POSTGRES_DSN"), "diagnostic requires guarded private PostgreSQL")
	previousRedis, previousBatch := common.RedisEnabled, common.BatchUpdateEnabled
	common.RedisEnabled, common.BatchUpdateEnabled = false, false
	t.Cleanup(func() { common.RedisEnabled, common.BatchUpdateEnabled = previousRedis, previousBatch })
	for _, source := range []string{"wallet", "subscription"} {
		for _, outcome := range []string{"exact", "lower", "refund-resize"} {
			t.Run(source+"/"+outcome, func(t *testing.T) {
				setupBillingLifecycleDB(t)
				seedUser(t, 801, 1000)
				seedToken(t, 801, 801, "synthetic-sql-diagnostic", 1000)
				requestID := "diagnostic-" + source + "-" + outcome
				info := &relaycommon.RelayInfo{UserId: 801, TokenId: 801, TokenKey: "synthetic-sql-diagnostic", RequestId: requestID, OriginModelName: "diagnostic-model", ForcePreConsume: true}
				var sub model.UserSubscription
				if source == "subscription" {
					plan := model.SubscriptionPlan{Title: "synthetic diagnostic", QuotaResetPeriod: model.SubscriptionResetNever}
					require.NoError(t, model.DB.Create(&plan).Error)
					model.InvalidateSubscriptionPlanCache(plan.Id)
					sub = model.UserSubscription{UserId: 801, PlanId: plan.Id, AmountTotal: 1000, Status: "active", ModelScope: "diagnostic-model", StartTime: time.Now().Unix() - 60, EndTime: time.Now().Unix() + 3600}
					require.NoError(t, model.DB.Create(&sub).Error)
					info.TokenUserSubscriptionId = sub.Id
				}
				diagnostic := &sqlWriteDiagnostic{Interface: logger.Default, counts: map[string]int{}}
				model.DB = model.DB.Session(&gorm.Session{Logger: diagnostic})
				c, _ := gin.CreateTestContext(nil)
				session, apiErr := NewBillingSession(c, info, 100)
				require.Nil(t, apiErr)
				diagnostic.flush(t, "preconsume")
				if outcome == "refund-resize" {
					require.NoError(t, session.Reserve(160))
					diagnostic.flush(t, "reserve")
					session.Refund(c)
					require.EventuallyWithT(t, func(collect *assert.CollectT) {
						var token model.Token
						assert.NoError(collect, model.DB.First(&token, 801).Error)
						assert.Equal(collect, 1000, token.RemainQuota)
						if source == "subscription" {
							var current model.UserSubscription
							assert.NoError(collect, model.DB.First(&current, sub.Id).Error)
							assert.Zero(collect, current.AmountUsed)
						} else {
							var user model.User
							assert.NoError(collect, model.DB.First(&user, 801).Error)
							assert.Equal(collect, 1000, user.Quota)
						}
						if os.Getenv("CILAI_DIAGNOSTIC_RECEIPTS") == "yes" {
							var receipt struct{ Status string }
							assert.NoError(collect, model.DB.Table("billing_refunds").Where("request_id = ?", requestID).Find(&receipt).Error)
							assert.Equal(collect, "succeeded", receipt.Status)
						}
					}, 5*time.Second, 10*time.Millisecond)
					diagnostic.flush(t, "refund")
				} else {
					actual := 100
					if outcome == "lower" {
						actual = 60
					}
					require.NoError(t, session.Settle(actual))
					diagnostic.flush(t, "settle")
					var token model.Token
					require.NoError(t, model.DB.First(&token, 801).Error)
					require.Equal(t, 1000-actual, token.RemainQuota)
					if source == "subscription" {
						var current model.UserSubscription
						require.NoError(t, model.DB.First(&current, sub.Id).Error)
						require.EqualValues(t, actual, current.AmountUsed)
					} else {
						var user model.User
						require.NoError(t, model.DB.First(&user, 801).Error)
						require.Equal(t, 1000-actual, user.Quota)
					}
				}
			})
		}
	}
}
